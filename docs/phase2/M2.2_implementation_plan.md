## 目标

实现**可追溯的证据状态机**，将面试中的每个回答链接到具体的Claim验证点，建立完整的证据链。

---

## 核心概念

### Evidence State Machine

```
UNSEEN (初始)
  ↓ [首次询问]
ADDRESSED (已询问)
  ↓ [有相关回答]
PARTIALLY_SUPPORTED (部分支持)
  ↓ [提供具体证据]
VERIFIED (已验证)

可能的异常路径:
- → UNSUPPORTED (不支持)
- → CONTRADICTORY (矛盾)
- → NEEDS_CLARIFICATION (需澄清)
```

### Key Features

1. **Evidence Span** - 定位到回答的具体文本段落
2. **Transition Tracking** - 记录每次状态变化
3. **Contradiction Detection** - 检测矛盾并生成澄清问题
4. **Policy Versioning** - 策略版本控制

```python
from enum import Enum
from dataclasses import dataclass

class EvidenceState(str, Enum):
    UNSEEN = "UNSEEN"
    ADDRESSED = "ADDRESSED"
    PARTIALLY_SUPPORTED = "PARTIALLY_SUPPORTED"
    VERIFIED = "VERIFIED"
    UNSUPPORTED = "UNSUPPORTED"
    CONTRADICTORY = "CONTRADICTORY"
    NEEDS_CLARIFICATION = "NEEDS_CLARIFICATION"

class ReasonCode(str, Enum):
    FIRST_INQUIRY = "FIRST_INQUIRY"
    HAS_ANSWER = "HAS_ANSWER"
    EVIDENCE_PROVIDED = "EVIDENCE_PROVIDED"
    EVIDENCE_SPANS_FOUND = "EVIDENCE_SPANS_FOUND"
    NO_EVIDENCE = "NO_EVIDENCE"
    CONTRADICTION_DETECTED = "CONTRADICTION_DETECTED"
    CLARIFICATION_NEEDED = "CLARIFICATION_NEEDED"
    CLARIFIED = "CLARIFIED"

@dataclass
class Transition:
    from_state: EvidenceState
    to_state: EvidenceState
    reason_code: ReasonCode
    guard: callable  # 转换条件

class EvidenceStateMachine:
    """Evidence state machine with guard conditions."""
    
    def __init__(self):
        self.transitions = self._define_transitions()
    
    def _define_transitions(self) -> list[Transition]:
        return [
            # UNSEEN → ADDRESSED
            Transition(
                EvidenceState.UNSEEN,
                EvidenceState.ADDRESSED,
                ReasonCode.FIRST_INQUIRY,
                guard=lambda ctx: ctx.has_question,
            ),
            
            # ADDRESSED → PARTIALLY_SUPPORTED
            Transition(
                EvidenceState.ADDRESSED,
                EvidenceState.PARTIALLY_SUPPORTED,
                ReasonCode.HAS_ANSWER,
                guard=lambda ctx: ctx.has_relevant_answer,
            ),
            
            # PARTIALLY_SUPPORTED → VERIFIED
            Transition(
                EvidenceState.PARTIALLY_SUPPORTED,
                EvidenceState.VERIFIED,
                ReasonCode.EVIDENCE_SPANS_FOUND,
                guard=lambda ctx: ctx.has_evidence_spans and ctx.strength >= 0.7,
            ),
            
            # ADDRESSED → UNSUPPORTED
            Transition(
                EvidenceState.ADDRESSED,
                EvidenceState.UNSUPPORTED,
                ReasonCode.NO_EVIDENCE,
                guard=lambda ctx: ctx.no_supporting_evidence,
            ),
            
            # PARTIALLY_SUPPORTED → CONTRADICTORY
            Transition(
                EvidenceState.PARTIALLY_SUPPORTED,
                EvidenceState.CONTRADICTORY,
                ReasonCode.CONTRADICTION_DETECTED,
                guard=lambda ctx: ctx.has_contradiction,
            ),
            
            # CONTRADICTORY → NEEDS_CLARIFICATION
            Transition(
                EvidenceState.CONTRADICTORY,
                EvidenceState.NEEDS_CLARIFICATION,
                ReasonCode.CLARIFICATION_NEEDED,
                guard=lambda ctx: ctx.clarification_question_generated,
            ),
            
            # NEEDS_CLARIFICATION → PARTIALLY_SUPPORTED
            Transition(
                EvidenceState.NEEDS_CLARIFICATION,
                EvidenceState.PARTIALLY_SUPPORTED,
                ReasonCode.CLARIFIED,
                guard=lambda ctx: ctx.clarification_provided,
            ),
        ]
    
    def can_transition(
        self,
        current_state: EvidenceState,
        target_state: EvidenceState,
        context: dict,
    ) -> tuple[bool, str | None]:
        """Check if transition is allowed.
        
        Returns:
            (allowed, reason_code_or_error)
        """
        for t in self.transitions:
            if t.from_state == current_state and t.to_state == target_state:
                if t.guard(context):
                    return True, t.reason_code.value
                else:
                    return False, f"Guard condition not met for {t.reason_code}"
        
        return False, f"No transition from {current_state} to {target_state}"
    
    def transition(
        self,
        verification_point_id: str,
        current_state: EvidenceState,
        target_state: EvidenceState,
        context: dict,
    ) -> tuple[bool, str]:
        """Execute state transition if allowed."""
        allowed, result = self.can_transition(current_state, target_state, context)
        
        if not allowed:
            return False, result
        
        # Record transition
        # ... (save to database)
        
        return True, result
```
